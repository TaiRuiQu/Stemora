cmake_minimum_required(VERSION 3.25)
project(StemoraInfra VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(STEMORA_GGML_SOURCE_DIR "" CACHE PATH "Pinned GGML source checkout")
set(STEMORA_GGML_BINARY_DIR "" CACHE PATH "Optional pinned prebuilt GGML DLLs and .lib import libraries")
option(STEMORA_ENABLE_CUDA "Build both CUDA and CPU; OFF creates CPU-only CI/test binaries" ON)
if(STEMORA_GGML_BINARY_DIR)
  set(_stemora_components ggml ggml-base ggml-cpu)
  if(STEMORA_ENABLE_CUDA)
    list(APPEND _stemora_components ggml-cuda)
  endif()
  foreach(_component IN LISTS _stemora_components)
    add_library(${_component} SHARED IMPORTED)
    set_target_properties(${_component} PROPERTIES
      IMPORTED_IMPLIB "${STEMORA_GGML_BINARY_DIR}/${_component}.lib"
      IMPORTED_LOCATION "${STEMORA_GGML_BINARY_DIR}/${_component}.dll"
      INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ggml/include")
  endforeach()
else()
if(NOT EXISTS "${STEMORA_GGML_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set STEMORA_GGML_SOURCE_DIR to the pinned GGML source. Use scripts/Build-Native.ps1.")
endif()
set(GGML_CUDA ${STEMORA_ENABLE_CUDA} CACHE BOOL "Build NVIDIA CUDA backend" FORCE)
set(GGML_CPU ON CACHE BOOL "Build CPU backend" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "Vulkan is not a supported Stemora backend" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Use explicit portable CPU instruction settings" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "No AVX512 requirement" FORCE)
set(GGML_AVX2 ON CACHE BOOL "AVX2 CPU kernels" FORCE)
set(GGML_AMX_TILE OFF CACHE BOOL "No AMX requirement" FORCE)
set(GGML_AMX_INT8 OFF CACHE BOOL "No AMX requirement" FORCE)
set(GGML_AMX_BF16 OFF CACHE BOOL "No AMX requirement" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "Use only compiled CPU/CUDA backends" FORCE)
set(GGML_CUDA_NO_VMM ON CACHE BOOL "Do not require nvcuda.dll for CPU-only operation" FORCE)
set(GGML_OPENMP ON CACHE BOOL "Parallel CPU inference" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "Stemora provides native tests" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "No extra examples" FORCE)
# GGML's global ccache launcher is unreliable with MSVC/Ninja on hosted runners.
set(GGML_CCACHE OFF CACHE BOOL "Use direct compiler invocations" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "Shared inference libraries" FORCE)
add_subdirectory("${STEMORA_GGML_SOURCE_DIR}" ggml EXCLUDE_FROM_ALL)
endif()
add_library(roformer_graph STATIC vendor/roformer/src/model.cpp vendor/roformer/src/inference.cpp vendor/roformer/src/utils.cpp)
target_include_directories(roformer_graph PUBLIC vendor/roformer/include PRIVATE vendor/roformer/src)
target_link_libraries(roformer_graph PUBLIC ggml ggml-base ggml-cpu)
if(STEMORA_ENABLE_CUDA)
  target_link_libraries(roformer_graph PUBLIC ggml-cuda)
endif()
target_compile_definitions(roformer_graph PUBLIC GGML_SHARED)
add_library(stemora_infra SHARED src/infra.cpp src/model_registry.cpp src/overlap_add.cpp src/json.cpp)
set_target_properties(stemora_infra PROPERTIES PREFIX "")
target_include_directories(stemora_infra PUBLIC include PRIVATE src)
target_compile_definitions(stemora_infra PRIVATE STEMORA_INFRA_BUILD)
target_link_libraries(stemora_infra PRIVATE roformer_graph)
add_executable(stemora-infra src/main.cpp src/json.cpp)
target_include_directories(stemora-infra PRIVATE include src)
target_link_libraries(stemora-infra PRIVATE stemora_infra)
if(MINGW)
  target_link_options(stemora-infra PRIVATE -municode -static-libgcc -static-libstdc++)
  target_link_options(stemora_infra PRIVATE -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(stemora-native-tests tests/infra_tests.cpp src/overlap_add.cpp src/json.cpp)
target_include_directories(stemora-native-tests PRIVATE include src)
target_link_libraries(stemora-native-tests PRIVATE stemora_infra)
if(MINGW)
  target_link_options(stemora-native-tests PRIVATE -static-libgcc -static-libstdc++)
endif()
add_test(NAME infra-unit-tests COMMAND stemora-native-tests)
